-- DROP vs TRUNCATE vs DELETE (test table) and AUTO_INCREMENT/rollback effects
mysql> CREATE TABLE ai_test (id INT AUTO_INCREMENT PRIMARY KEY, note VARCHAR(20)) ENGINE=InnoDB;
mysql> INSERT INTO ai_test (note) VALUES ('a'),('b'),('c'); -- ids 1,2,3
mysql> START TRANSACTION;
mysql> DELETE FROM ai_test; -- DML, row-by-row, logs rows
mysql> ROLLBACK;
mysql> SELECT COUNT(*) AS cnt, MAX(id) AS max_id FROM ai_test;
+-----+--------+
| cnt | max_id |
+-----+--------+
|   3 |      3 |
+-----+--------+
-- DELETE can be rolled back (within transaction) and does NOT reset AUTO_INCREMENT (next id would be 4).

mysql> TRUNCATE TABLE ai_test; -- DDL-like, drops/recreates storage
mysql> INSERT INTO ai_test (note) VALUES ('x');
mysql> SELECT id FROM ai_test;
+----+
| id |
+----+
|  1 |
+----+
-- TRUNCATE resets AUTO_INCREMENT counter; implicit commit in MySQL (cannot be rolled back as row restore).

mysql> DROP TABLE ai_test;
-- DROP removes table definition and data entirely (and its AUTO_INCREMENT metadata).
