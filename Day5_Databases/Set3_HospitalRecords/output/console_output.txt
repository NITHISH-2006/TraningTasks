Query OK, 0 rows affected (table renamed within database)
mysql> DROP DATABASE hospital_admin_demo;
Query OK (database and its tables removed)
mysql> CREATE DATABASE hospital_admin_demo;
Query OK, 1 row affected (recreated empty database)
-- Effect on files (MySQL 8.4/InnoDB): CREATE DATABASE creates a directory in datadir; CREATE TABLE creates .ibd (file-per-table) and metadata in data dictionary; RENAME updates metadata/names; DROP DATABASE removes the schema directory and contained InnoDB tablespace files; recreate yields a new empty schema directory.

-- Retrieve patients visited in a given month (e.g. October 2026)
mysql> SELECT p.patient_id, p.full_name, v.visit_date, v.diagnosis
    -> FROM patients p JOIN visits v ON p.patient_id = v.patient_id
    -> WHERE DATE_FORMAT(v.visit_date,'%Y-%m') = '2026-10'
    -> ORDER BY v.visit_date;
